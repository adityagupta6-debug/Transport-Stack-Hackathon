import { ChangeEvent, useMemo, useRef, useState } from 'react';
import { Download, RotateCcw, Search, Upload } from 'lucide-react';
import { DELHI_ROUTES } from '../data/delhiTransitData';
import { CrowdingLevel, DatasetPipelineStats, DayType, TransportStackRecord } from '../types/transit';
import { exportOctoberDelhiSampleCsv } from '../utils/transitAnalytics';
import { CrowdingIndicator } from './CrowdingIndicator';

interface DatasetPipelineViewProps {
  records: TransportStackRecord[];
  stats: DatasetPipelineStats;
  passUpliftFactor: number;
  onChangePassUplift: (factor: number) => void;
  onUploadCsv: (csvText: string, fileName: string) => void;
  onResetBaseline: () => void;
  uploadError: string | null;
}

export function DatasetPipelineView({
  records,
  stats,
  passUpliftFactor,
  onChangePassUplift,
  onUploadCsv,
  onResetBaseline,
  uploadError,
}: DatasetPipelineViewProps) {
  const fileInputRef = useRef<HTMLInputElement>(null);
  const [searchQuery, setSearchQuery] = useState('');
  const [routeFilter, setRouteFilter] = useState('all');
  const [dayFilter, setDayFilter] = useState<DayType>('weekday');
  const [crowdingFilter, setCrowdingFilter] = useState<'all' | CrowdingLevel>('all');
  const [hourFilter, setHourFilter] = useState<string>('18');
  const [imgError, setImgError] = useState(false);

  const filteredRecords = useMemo(() => {
    const q = searchQuery.trim().toLowerCase();
    return records
      .filter((r) => {
        if (r.dayType !== dayFilter) return false;
        if (routeFilter !== 'all' && r.routeId !== routeFilter) return false;
        if (crowdingFilter !== 'all' && r.crowdingLevel !== crowdingFilter) return false;
        if (hourFilter !== 'all' && r.hour !== Number(hourFilter)) return false;
        if (
          q &&
          !r.stopName.toLowerCase().includes(q) &&
          !r.routeNumber.toLowerCase().includes(q) &&
          !r.busId.toLowerCase().includes(q)
        ) {
          return false;
        }
        return true;
      })
      .slice(0, 60);
  }, [records, dayFilter, routeFilter, crowdingFilter, hourFilter, searchQuery]);

  const handleFileChange = (e: ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = () => {
      if (typeof reader.result === 'string') {
        onUploadCsv(reader.result, file.name);
      }
    };
    reader.readAsText(file);
    e.target.value = '';
  };

  const handleDownloadSample = () => {
    const csv = exportOctoberDelhiSampleCsv();
    const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = 'October-Delhi-sample.csv';
    a.click();
    URL.revokeObjectURL(url);
  };

  return (
    <div className="space-y-6">
      {/* Reproducible Pipeline & Fleet Banner */}
      <section className="bg-white border border-slate-200 rounded-lg overflow-hidden grid grid-cols-1 lg:grid-cols-12 items-stretch">
        <div className="lg:col-span-7 p-6 flex flex-col justify-between gap-4">
          <div>
            <div className="text-xs font-semibold text-blue-700">
              Reproducible Ticket-to-Crowding Pipeline
            </div>
            <h2 className="text-xl font-bold text-slate-900 mt-1">
              From October-Delhi.csv &amp; Static GTFS to Stretch Load
            </h2>
            <p className="text-sm text-slate-600 mt-2 leading-relaxed max-w-2xl">
              Ticket rows (<span className="font-mono text-xs">passenger_count, vehicle_number, start_stop_name, end_stop_name, route, created_at, is_ac</span>) are joined to static GTFS stop sequences, grouped by bus trip, and converted into running on-board passenger counts with pass-holder uplift and 12m/9m capacity normalization.
            </p>
          </div>

          <div className="grid grid-cols-1 sm:grid-cols-3 gap-4 pt-3 border-t border-slate-100">
            <div>
              <div className="text-xs font-semibold text-slate-900">01. Trip Grouping</div>
              <p className="text-xs text-slate-500 mt-0.5">
                Clusters tickets by <span className="font-mono">vehicle_number + route</span> into scheduled GTFS trip windows.
              </p>
            </div>
            <div>
              <div className="text-xs font-semibold text-slate-900">02. Running Count</div>
              <p className="text-xs text-slate-500 mt-0.5">
                Accumulates <span className="font-mono">+boardings - alightings</span> along every directional stop stretch.
              </p>
            </div>
            <div>
              <div className="text-xs font-semibold text-slate-900">03. Pass &amp; Size Scaling</div>
              <p className="text-xs text-slate-500 mt-0.5">
                Applies un-ticketed pass uplift and scales by 12m (68–74 cap) vs 9m (45 cap) bus size.
              </p>
            </div>
          </div>

          <div className="flex flex-wrap items-center gap-3 pt-2">
            <input
              ref={fileInputRef}
              type="file"
              accept=".csv,text/csv"
              onChange={handleFileChange}
              className="hidden"
            />
            <button
              type="button"
              onClick={() => fileInputRef.current?.click()}
              className="inline-flex items-center gap-2 px-4 py-2 text-xs font-semibold text-white bg-slate-900 hover:bg-slate-800 rounded-md transition-colors whitespace-nowrap"
            >
              <Upload className="w-3.5 h-3.5" />
              Upload October-Delhi.csv
            </button>
            <button
              type="button"
              onClick={handleDownloadSample}
              className="inline-flex items-center gap-2 px-4 py-2 text-xs font-semibold text-slate-700 bg-white border border-slate-300 hover:bg-slate-50 rounded-md transition-colors whitespace-nowrap"
            >
              <Download className="w-3.5 h-3.5" />
              Download Schema Sample CSV
            </button>
            {stats.isCustomDataset && (
              <button
                type="button"
                onClick={onResetBaseline}
                className="inline-flex items-center gap-1.5 px-3 py-2 text-xs font-medium text-slate-600 hover:text-slate-900 transition-colors whitespace-nowrap"
              >
                <RotateCcw className="w-3.5 h-3.5" />
                Reset to October Baseline
              </button>
            )}
          </div>

          {uploadError && (
            <div className="text-xs text-red-700 bg-red-50 border border-red-200 rounded-md px-3 py-2">
              {uploadError}
            </div>
          )}
        </div>

        <div className="lg:col-span-5 relative min-h-[220px] bg-slate-100 border-t lg:border-t-0 lg:border-l border-slate-200">
          {!imgError ? (
            <img
              src="/src/assets/images/delhi_transit_corridor_1795510191236.jpg"
              alt="Delhi electric low-floor public buses operating along a central transit corridor"
              referrerPolicy="no-referrer"
              onError={() => setImgError(true)}
              className="w-full h-full object-cover"
            />
          ) : (
            <div className="w-full h-full flex flex-col items-center justify-center p-6 text-center bg-slate-900 text-slate-300">
              <span className="text-sm font-semibold text-white">Delhi DTC Electric Fleet</span>
              <span className="text-xs text-slate-400 mt-1">
                Static GTFS + ETM &amp; E-Ticketing Pipeline
              </span>
            </div>
          )}
        </div>
      </section>

      {/* Pipeline Summary Stats */}
      <section className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-6 gap-3">
        {[
          { label: 'Active Dataset', value: stats.sourceLabel, isText: true },
          { label: 'Coverage Window', value: stats.dateRange, isText: true },
          { label: 'Ticket Records', value: stats.rawTicketRows.toLocaleString() },
          { label: 'GTFS Trip Buckets', value: stats.gtfsMatchedTrips.toLocaleString() },
          { label: 'Low-Sample Cells', value: stats.lowSampleCells.toLocaleString() },
          { label: 'Stretch Estimates', value: stats.cleanedStretchRecords.toLocaleString() },
        ].map((item) => (
          <div key={item.label} className="bg-white border border-slate-200 rounded-lg p-3.5">
            <div className="text-[11px] font-medium text-slate-500">{item.label}</div>
            <div
              className={`mt-1 font-bold text-slate-900 truncate ${
                item.isText ? 'text-xs' : 'text-lg font-mono tabular-nums'
              }`}
              title={String(item.value)}
            >
              {item.value}
            </div>
          </div>
        ))}
      </section>

      {/* Technical Note: Method, Assumptions, Pass-Holder Uplift & Reliability Limits */}
      <section className="bg-white border border-slate-200 rounded-lg p-6 space-y-5">
        <div className="flex flex-col lg:flex-row lg:items-center justify-between gap-4 pb-4 border-b border-slate-100">
          <div>
            <h3 className="text-base font-bold text-slate-900">
              Technical Note: Model Assumptions &amp; Where Estimates Are Least Reliable
            </h3>
            <p className="text-xs text-slate-500 mt-0.5">
              Transparent documentation of ticket-to-load modeling choices and sensitivity controls
            </p>
          </div>

          <div className="flex items-center gap-3 bg-slate-50 border border-slate-200 rounded-md px-3.5 py-2">
            <label htmlFor="uplift-slider" className="text-xs font-semibold text-slate-700 whitespace-nowrap">
              Pass-Holder Uplift:
            </label>
            <input
              id="uplift-slider"
              type="range"
              min={1.0}
              max={1.4}
              step={0.02}
              value={passUpliftFactor}
              onChange={(e) => onChangePassUplift(Number(e.target.value))}
              className="w-28 accent-blue-700 cursor-pointer"
            />
            <span className="text-xs font-mono font-bold text-blue-700 tabular-nums whitespace-nowrap">
              {passUpliftFactor.toFixed(2)}&times; (+{Math.round((passUpliftFactor - 1) * 100)}%)
            </span>
          </div>
        </div>

        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-5 text-xs leading-relaxed">
          <div>
            <h4 className="font-bold text-slate-900 mb-1">
              1. Accounting for Pass Holders &amp; Un-ticketed Riders
            </h4>
            <p className="text-slate-600">
              Monthly DTC/Cluster pass holders, student passes, and zero-fare riders do not generate paid ETM or QR transactions. We apply a calibrated multiplier (default <strong>1.22&times;</strong>, adjustable above) to paid boardings so predicted loads do not systematically undercount peak commuter buses.
            </p>
          </div>

          <div>
            <h4 className="font-bold text-slate-900 mb-1">
              2. Planned vs. Actual Alighting Stop
            </h4>
            <p className="text-slate-600">
              Tickets record <span className="font-mono">end_stop_name</span> where passengers <em>intend</em> to exit. Actual alighting can shift by ±1 stop near major markets or metro interchanges. Stretch tables display a ±11% dispersion band around the running count.
            </p>
          </div>

          <div>
            <h4 className="font-bold text-slate-900 mb-1">
              3. Bus Size Normalization (12m vs. 9m)
            </h4>
            <p className="text-slate-600">
              Raw passenger counts cannot be compared directly across fleets: 40 passengers leave standing room on a <strong>12m Standard</strong> bus (38–40 seats, 68–74 max), whereas 40 passengers represent <strong>89% Crowded</strong> density on a <strong>9m Midi</strong> feeder (26 seats, 45 max).
            </p>
          </div>

          <div>
            <h4 className="font-bold text-slate-900 mb-1">
              4. Where Estimates Are Least Reliable
            </h4>
            <p className="text-slate-600">
              Estimates are least reliable on low-frequency feeder hours (before 06:30 and after 21:30, marked <strong>Low confidence</strong> when sample trips &lt; 25), on trips with missing ETM stop-code alignment, and during unmodeled seasonal shocks (monsoon, festivals) outside the 31-day October window.
            </p>
          </div>
        </div>
      </section>

      {/* Cleaned Stretch Records Explorer */}
      <section className="bg-white border border-slate-200 rounded-lg p-5">
        <div className="flex flex-col lg:flex-row lg:items-center justify-between gap-4 mb-4">
          <div>
            <h3 className="text-base font-bold text-slate-900">
              Queryable Stretch-Level Crowding Outputs
            </h3>
            <p className="text-xs text-slate-500">
              Filter stretch crowding estimates by route, direction, day type, hour, or confidence
            </p>
          </div>

          <div className="flex flex-wrap items-center gap-2.5">
            <div className="relative">
              <Search className="w-3.5 h-3.5 text-slate-400 absolute left-2.5 top-1/2 -translate-y-1/2" />
              <input
                type="text"
                value={searchQuery}
                onChange={(e) => setSearchQuery(e.target.value)}
                placeholder="Search stop, route, vehicle..."
                className="h-9 pl-8 pr-3 text-xs border border-slate-300 rounded-md bg-white text-slate-900 focus:outline-none focus:ring-2 focus:ring-blue-600"
              />
            </div>

            <select
              value={dayFilter}
              onChange={(e) => setDayFilter(e.target.value as DayType)}
              aria-label="Filter by day type"
              className="h-9 px-2.5 text-xs border border-slate-300 rounded-md bg-white text-slate-900"
            >
              <option value="weekday">Weekday</option>
              <option value="weekend">Weekend</option>
            </select>

            <select
              value={routeFilter}
              onChange={(e) => setRouteFilter(e.target.value)}
              aria-label="Filter by route"
              className="h-9 px-2.5 text-xs border border-slate-300 rounded-md bg-white text-slate-900"
            >
              <option value="all">All Routes</option>
              {DELHI_ROUTES.map((r) => (
                <option key={r.id} value={r.id}>
                  {r.routeNumber} ({r.busSize})
                </option>
              ))}
            </select>

            <select
              value={hourFilter}
              onChange={(e) => setHourFilter(e.target.value)}
              aria-label="Filter by hour"
              className="h-9 px-2.5 text-xs border border-slate-300 rounded-md bg-white text-slate-900"
            >
              <option value="all">All Hours</option>
              {Array.from({ length: 18 }, (_, i) => i + 5).map((h) => (
                <option key={h} value={String(h)}>
                  {String(h).padStart(2, '0')}:00
                </option>
              ))}
            </select>

            <select
              value={crowdingFilter}
              onChange={(e) => setCrowdingFilter(e.target.value as 'all' | CrowdingLevel)}
              aria-label="Filter by crowding level"
              className="h-9 px-2.5 text-xs border border-slate-300 rounded-md bg-white text-slate-900"
            >
              <option value="all">All Crowding Levels</option>
              <option value="Seats available">Seats available</option>
              <option value="Standing room">Standing room</option>
              <option value="Crowded">Crowded</option>
            </select>
          </div>
        </div>

        {filteredRecords.length === 0 ? (
          <div className="py-10 text-center border-t border-slate-100">
            <p className="text-sm font-semibold text-slate-900">
              No stretch estimates match the active filters
            </p>
            <button
              type="button"
              onClick={() => {
                setSearchQuery('');
                setRouteFilter('all');
                setHourFilter('all');
                setCrowdingFilter('all');
              }}
              className="mt-2 text-xs font-semibold text-blue-700 hover:underline"
            >
              Clear filters
            </button>
          </div>
        ) : (
          <div className="overflow-x-auto">
            <table className="w-full text-xs">
              <thead>
                <tr className="text-left text-[11px] text-slate-500 border-b border-slate-200">
                  <th className="py-2 pr-3 font-medium">Hour</th>
                  <th className="py-2 px-2 font-medium">Route</th>
                  <th className="py-2 px-2 font-medium">Bus Size</th>
                  <th className="py-2 px-2 font-medium">Stop Stretch</th>
                  <th className="py-2 px-2 font-medium text-right">Paid On</th>
                  <th className="py-2 px-2 font-medium text-right">+Passes</th>
                  <th className="py-2 px-2 font-medium text-right">On-Board</th>
                  <th className="py-2 px-2 font-medium text-right">Trips (Conf.)</th>
                  <th className="py-2 pl-2 font-medium text-right">Crowding</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-100">
                {filteredRecords.map((r) => (
                  <tr key={r.id} className="hover:bg-slate-50">
                    <td className="py-2 pr-3 font-mono text-slate-700 tabular-nums">
                      {r.timestamp}
                    </td>
                    <td className="py-2 px-2 font-mono font-semibold text-slate-900">
                      {r.routeNumber}
                    </td>
                    <td className="py-2 px-2 text-slate-600 whitespace-nowrap">
                      {r.busSize}
                    </td>
                    <td className="py-2 px-2 text-slate-800">{r.stopName}</td>
                    <td className="py-2 px-2 text-right font-mono text-slate-700 tabular-nums">
                      {r.paidBoardings}
                    </td>
                    <td className="py-2 px-2 text-right font-mono text-blue-700 tabular-nums">
                      {r.totalBoardingsWithPasses}
                    </td>
                    <td className="py-2 px-2 text-right font-mono text-slate-800 tabular-nums">
                      {r.cumulativeLoad}/{r.crushCapacity}
                    </td>
                    <td className="py-2 px-2 text-right font-mono text-slate-600 tabular-nums">
                      {r.sampleTrips} ({r.confidence})
                    </td>
                    <td className="py-2 pl-2 text-right">
                      <CrowdingIndicator
                        level={r.crowdingLevel}
                        loadPercentage={r.loadPercentage}
                        compact
                      />
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        )}
      </section>
    </div>
  );
}
