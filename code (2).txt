import { useMemo, useState } from 'react';
import { DELHI_ROUTES, WEEKLY_STABILITY_DATA } from '../data/delhiTransitData';
import { DatasetPipelineStats, DayType, TransportStackRecord } from '../types/transit';
import {
  buildHourlyDemandSeries,
  buildRouteHourHeatmap,
  computeOperatorDispatchPriorities,
} from '../utils/transitAnalytics';
import { CrowdingIndicator } from './CrowdingIndicator';

interface DemandAnalyticsViewProps {
  records: TransportStackRecord[];
  stats: DatasetPipelineStats;
  onInspectCorridorInPlanner: (
    originId: string,
    destinationId: string,
    time: string,
    dayType: DayType
  ) => void;
}

export function DemandAnalyticsView({
  records,
  stats,
  onInspectCorridorInPlanner,
}: DemandAnalyticsViewProps) {
  const [routeFilter, setRouteFilter] = useState<string>('all');
  const [heatmapDayType, setHeatmapDayType] = useState<DayType>('weekday');
  const [heatmapDirFilter, setHeatmapDirFilter] = useState<'all' | 'UP_CW' | 'DOWN_ACW'>('all');
  const [selectedHour, setSelectedHour] = useState<number>(18);
  const [hoveredHour, setHoveredHour] = useState<number | null>(null);
  const [selectedHeatmapCell, setSelectedHeatmapCell] = useState<{
    routeId: string;
    routeNumber: string;
    routeName: string;
    hour: number;
    peakLoad: number;
    avgLoad: number;
    avgSpeed: number;
    sampleTrips: number;
  } | null>(null);

  const hourlySeries = useMemo(
    () => buildHourlyDemandSeries(records, routeFilter),
    [records, routeFilter]
  );

  const heatmapRows = useMemo(
    () => buildRouteHourHeatmap(records, heatmapDayType, heatmapDirFilter),
    [records, heatmapDayType, heatmapDirFilter]
  );

  const dispatchPriorities = useMemo(
    () => computeOperatorDispatchPriorities(records, heatmapDayType),
    [records, heatmapDayType]
  );

  const kpis = useMemo(() => {
    const weekdayRecs = records.filter((r) => r.dayType === 'weekday');
    const weekendRecs = records.filter((r) => r.dayType === 'weekend');
    const weekdayMean =
      weekdayRecs.length > 0
        ? Math.round(
            weekdayRecs.reduce((s, r) => s + r.loadPercentage, 0) / weekdayRecs.length
          )
        : 0;
    const weekendMean =
      weekendRecs.length > 0
        ? Math.round(
            weekendRecs.reduce((s, r) => s + r.loadPercentage, 0) / weekendRecs.length
          )
        : 0;

    const peakHourPoint = [...hourlySeries].sort((a, b) => b.totalVolume - a.totalVolume)[0];
    const overloadedHours = weekdayRecs.filter((r) => r.crowdingLevel === 'Crowded').length;
    const avgWeeklyMape = (
      WEEKLY_STABILITY_DATA.reduce((s, w) => s + w.mapeVsBaselinePct, 0) /
      WEEKLY_STABILITY_DATA.length
    ).toFixed(1);

    return {
      weekdayMean,
      weekendMean,
      peakHourLabel: peakHourPoint?.label ?? '18:00',
      overloadedShare:
        weekdayRecs.length > 0
          ? Math.round((overloadedHours / weekdayRecs.length) * 100)
          : 0,
      avgWeeklyMape,
    };
  }, [records, hourlySeries]);

  const activeRouteForStopProfile = useMemo(() => {
    if (routeFilter !== 'all') {
      return DELHI_ROUTES.find((r) => r.id === routeFilter) ?? DELHI_ROUTES[0];
    }
    if (selectedHeatmapCell) {
      return DELHI_ROUTES.find((r) => r.id === selectedHeatmapCell.routeId) ?? DELHI_ROUTES[0];
    }
    return DELHI_ROUTES[0];
  }, [routeFilter, selectedHeatmapCell]);

  const stopProfileRecords = useMemo(() => {
    const hr = selectedHeatmapCell?.hour ?? selectedHour;
    return records
      .filter(
        (r) =>
          r.dayType === heatmapDayType &&
          r.routeId === activeRouteForStopProfile.id &&
          r.hour === hr
      )
      .sort((a, b) => a.stopSequence - b.stopSequence);
  }, [records, heatmapDayType, activeRouteForStopProfile, selectedHeatmapCell, selectedHour]);

  const activeHoverPoint =
    hourlySeries.find((p) => p.hour === (hoveredHour ?? selectedHour)) ?? hourlySeries[13];

  return (
    <div className="space-y-6">
      {/* 4-Column Validation & Peak KPIs */}
      <section className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
        <div className="bg-white border border-slate-200 rounded-lg p-4">
          <div className="text-xs font-medium text-slate-500">
            {stats.isCustomDataset ? 'Uploaded Dataset Scope' : 'October 2026 Dataset Scope'}
          </div>
          <div className="mt-1.5 text-2xl font-bold font-mono text-slate-900 tabular-nums">
            {stats.rawTicketRows >= 1_000_000
              ? `${(stats.rawTicketRows / 1_000_000).toFixed(2)}M tickets`
              : `${stats.rawTicketRows.toLocaleString()} tickets`}
          </div>
          <div className="mt-1 text-xs text-slate-500 truncate" title={`${stats.dateRange} · ${stats.sourceLabel}`}>
            {stats.dateRange} · {stats.gtfsMatchedTrips.toLocaleString()} trip buckets
          </div>
        </div>

        <div className="bg-white border border-slate-200 rounded-lg p-4">
          <div className="text-xs font-medium text-slate-500">
            Weekday vs. Weekend Mean Load
          </div>
          <div className="mt-1.5 text-2xl font-bold font-mono text-slate-900 tabular-nums">
            {kpis.weekdayMean}% vs {kpis.weekendMean}%
          </div>
          <div className="mt-1 text-xs text-slate-500">
            Peak demand window: {kpis.peakHourLabel} IST
          </div>
        </div>

        <div className="bg-white border border-slate-200 rounded-lg p-4">
          <div className="text-xs font-medium text-slate-500">
            Week-over-Week Model Stability
          </div>
          <div className="mt-1.5 text-2xl font-bold font-mono text-emerald-700 tabular-nums">
            {kpis.avgWeeklyMape}% MAPE
          </div>
          <div className="mt-1 text-xs text-slate-500">
            Weeks 1–3 calibration vs. Week 4 holdout
          </div>
        </div>

        <div className="bg-white border border-slate-200 rounded-lg p-4">
          <div className="text-xs font-medium text-slate-500">
            Overloaded Weekday Stretches
          </div>
          <div className="mt-1.5 text-2xl font-bold font-mono text-red-700 tabular-nums">
            {kpis.overloadedShare}%
          </div>
          <div className="mt-1 text-xs text-slate-500">
            Corridor-hours classified as Crowded (&gt;85%)
          </div>
        </div>
      </section>

      {/* Weekday vs Weekend Hourly Demand & Load Validation Chart */}
      <section className="bg-white border border-slate-200 rounded-lg p-5">
        <div className="flex flex-col md:flex-row md:items-center justify-between gap-4 mb-4">
          <div>
            <h2 className="text-base font-bold text-slate-900">
              Known Pattern Check: Morning &amp; Evening Peaks (Weekday vs. Weekend)
            </h2>
            <p className="text-xs text-slate-500 mt-0.5">
              Compares Weekday AM (08:30–10:30) and PM (17:30–19:30) surges against flatter Weekend demand
            </p>
          </div>

          <div className="flex flex-wrap items-center gap-3">
            <label htmlFor="analytics-route-filter" className="text-xs font-medium text-slate-600">
              Route &amp; Direction:
            </label>
            <select
              id="analytics-route-filter"
              value={routeFilter}
              onChange={(e) => setRouteFilter(e.target.value)}
              className="h-9 px-3 text-xs bg-white border border-slate-300 rounded-md text-slate-900 focus:outline-none focus:ring-2 focus:ring-blue-600"
            >
              <option value="all">All Network Corridors</option>
              {DELHI_ROUTES.map((r) => (
                <option key={r.id} value={r.id}>
                  {r.routeNumber} · {r.name} ({r.busSize})
                </option>
              ))}
            </select>
          </div>
        </div>

        {activeHoverPoint && (
          <div className="mb-4 px-4 py-2.5 bg-slate-50 border border-slate-200 rounded-md flex flex-wrap items-center justify-between gap-4 text-xs">
            <div className="flex flex-wrap items-center gap-4">
              <span className="font-mono font-bold text-slate-900 tabular-nums">
                {activeHoverPoint.label} IST
              </span>
              <span className="text-slate-600 tabular-nums">
                Weekday Load:{' '}
                <strong className="font-mono text-red-700">
                  {activeHoverPoint.avgLoadPercentage}%
                </strong>
              </span>
              <span className="text-slate-600 tabular-nums">
                Weekend Load:{' '}
                <strong className="font-mono text-teal-700">
                  {activeHoverPoint.weekendAvgLoadPercentage}%
                </strong>
              </span>
              <span className="text-slate-600 tabular-nums">
                ETM + E-Ticket Volume:{' '}
                <strong className="font-mono text-slate-900">
                  {activeHoverPoint.totalVolume}
                </strong>
              </span>
            </div>
            <CrowdingIndicator
              level={activeHoverPoint.crowdingLevel}
              loadPercentage={activeHoverPoint.avgLoadPercentage}
              compact
            />
          </div>
        )}

        <WeekdayVsWeekendChart
          data={hourlySeries}
          selectedHour={selectedHour}
          onHoverHour={setHoveredHour}
          onSelectHour={(hr) => setSelectedHour(hr)}
        />
      </section>

      {/* Operator Dispatch Priorities + Week-over-Week Consistency Check */}
      <div className="grid grid-cols-1 xl:grid-cols-12 gap-6 items-start">
        {/* Bus Operator View: Where extra buses help most */}
        <section className="xl:col-span-7 bg-white border border-slate-200 rounded-lg p-5">
          <div className="flex flex-wrap items-center justify-between gap-2 mb-4">
            <div>
              <h3 className="text-base font-bold text-slate-900">
                Bus Operator View: Overloaded Corridors Needing Extra Buses
              </h3>
              <p className="text-xs text-slate-500">
                Ranked by peak stretch load above bus capacity ({heatmapDayType})
              </p>
            </div>
          </div>

          <div className="overflow-x-auto">
            <table className="w-full text-xs">
              <thead>
                <tr className="text-left text-[11px] text-slate-500 border-b border-slate-200">
                  <th className="py-2 pr-3 font-medium">Route</th>
                  <th className="py-2 px-2 font-medium">Bottleneck Corridor</th>
                  <th className="py-2 px-2 font-medium">Hour</th>
                  <th className="py-2 px-2 font-medium">Size</th>
                  <th className="py-2 px-2 font-medium text-right">Peak Load</th>
                  <th className="py-2 pl-2 font-medium">Dispatch Recommendation</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-100">
                {dispatchPriorities.map((item) => (
                  <tr key={`${item.routeId}-${item.hour}`} className="hover:bg-slate-50">
                    <td className="py-2.5 pr-3 font-mono font-bold text-slate-900">
                      {item.routeLabel}
                    </td>
                    <td className="py-2.5 px-2 text-slate-800">{item.corridor}</td>
                    <td className="py-2.5 px-2 font-mono text-slate-700 tabular-nums">
                      {String(item.hour).padStart(2, '0')}:00
                    </td>
                    <td className="py-2.5 px-2 text-slate-600 whitespace-nowrap">
                      {item.busSize}
                    </td>
                    <td className="py-2.5 px-2 text-right font-mono font-bold text-red-700 tabular-nums">
                      {item.peakLoadPct}% (+{item.excessPaxPerBus} standing)
                    </td>
                    <td className="py-2.5 pl-2 text-slate-700">{item.recommendedAction}</td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </section>

        {/* Week-over-Week Validation (Oct 1-31, 2026) */}
        <section className="xl:col-span-5 bg-white border border-slate-200 rounded-lg p-5">
          <div className="mb-4">
            <h3 className="text-base font-bold text-slate-900">
              Week-over-Week Reliability (Oct 1–31, 2026)
            </h3>
            <p className="text-xs text-slate-500 mt-0.5">
              Checking how crowding estimates hold up from one week to the next across 4.5 weeks
            </p>
          </div>

          <div className="overflow-x-auto">
            <table className="w-full text-xs">
              <thead>
                <tr className="text-left text-[11px] text-slate-500 border-b border-slate-200">
                  <th className="py-2 pr-2 font-medium">Window</th>
                  <th className="py-2 px-2 font-medium text-right">Tickets</th>
                  <th className="py-2 px-2 font-medium text-right">AM Peak</th>
                  <th className="py-2 px-2 font-medium text-right">PM Peak</th>
                  <th className="py-2 pl-2 font-medium text-right">Error (MAPE)</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-100">
                {WEEKLY_STABILITY_DATA.map((w) => (
                  <tr key={w.weekLabel}>
                    <td className="py-2.5 pr-2">
                      <div className="font-semibold text-slate-900">{w.weekLabel}</div>
                      <div className="text-[11px] text-slate-500">{w.dateRange}</div>
                    </td>
                    <td className="py-2.5 px-2 text-right font-mono text-slate-700 tabular-nums">
                      {(w.ticketCount / 1000).toFixed(0)}k
                    </td>
                    <td className="py-2.5 px-2 text-right font-mono text-slate-800 tabular-nums">
                      {w.amPeakLoadPct}%
                    </td>
                    <td className="py-2.5 px-2 text-right font-mono text-slate-800 tabular-nums">
                      {w.pmPeakLoadPct}%
                    </td>
                    <td className="py-2.5 pl-2 text-right font-mono font-semibold text-emerald-700 tabular-nums">
                      {w.mapeVsBaselinePct}%
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>

          <p className="mt-3 pt-3 border-t border-slate-100 text-xs text-slate-600 leading-relaxed">
            Weekly peak load varies by &lt;4% across normal October weekdays, confirming that 4–5 weeks reliably captures commuting rhythms—though it does not capture monsoon or major festival shocks.
          </p>
        </section>
      </div>

      {/* Route x Hour Crowding Heatmap + Stop Load Inspector */}
      <div className="grid grid-cols-1 xl:grid-cols-12 gap-6 items-start">
        <section className="xl:col-span-7 bg-white border border-slate-200 rounded-lg p-5">
          <div className="flex flex-wrap items-center justify-between gap-3 mb-4">
            <div>
              <h3 className="text-base font-bold text-slate-900">
                Route &times; Hour Crowding Matrix
              </h3>
              <p className="text-xs text-slate-500">
                Click any cell to inspect stop-by-stop load and sample trip confidence
              </p>
            </div>

            <div className="flex flex-wrap items-center gap-2">
              <div className="flex items-center gap-1 p-1 bg-slate-100 rounded-md">
                {(
                  [
                    { id: 'all', label: 'All Routes' },
                    { id: 'UP_CW', label: 'UP / CW' },
                    { id: 'DOWN_ACW', label: 'DOWN / ACW' },
                  ] as const
                ).map((d) => (
                  <button
                    key={d.id}
                    type="button"
                    onClick={() => setHeatmapDirFilter(d.id)}
                    className={`px-2.5 py-1 text-xs font-medium rounded transition-colors ${
                      heatmapDirFilter === d.id
                        ? 'bg-white text-slate-900 shadow-xs'
                        : 'text-slate-600 hover:text-slate-900'
                    }`}
                  >
                    {d.label}
                  </button>
                ))}
              </div>
              <div className="flex items-center gap-1 p-1 bg-slate-100 rounded-md">
                {(['weekday', 'weekend'] as const).map((dt) => (
                  <button
                    key={dt}
                    type="button"
                    onClick={() => setHeatmapDayType(dt)}
                    className={`px-3 py-1 text-xs font-medium rounded transition-colors ${
                      heatmapDayType === dt
                        ? 'bg-white text-slate-900 shadow-xs'
                        : 'text-slate-600 hover:text-slate-900'
                    }`}
                  >
                    {dt === 'weekday' ? 'Weekday' : 'Weekend'}
                  </button>
                ))}
              </div>
            </div>
          </div>

          <div className="overflow-x-auto">
            <table className="w-full border-collapse text-[11px]">
              <thead>
                <tr className="text-slate-500">
                  <th className="py-1.5 pr-3 text-left font-medium whitespace-nowrap">Route</th>
                  {Array.from({ length: 17 }, (_, i) => i + 6).map((hr) => (
                    <th
                      key={hr}
                      className="py-1.5 px-0.5 text-center font-mono font-normal tabular-nums"
                    >
                      {String(hr).padStart(2, '0')}
                    </th>
                  ))}
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-100">
                {heatmapRows.map(({ route, hours }) => (
                  <tr key={route.id}>
                    <td className="py-1.5 pr-3 font-mono font-semibold text-slate-800 whitespace-nowrap">
                      {route.routeNumber}
                    </td>
                    {hours.map((cell) => {
                      const bg =
                        cell.level === 'Seats available'
                          ? 'bg-emerald-100 text-emerald-900 hover:bg-emerald-200'
                          : cell.level === 'Standing room'
                          ? 'bg-amber-200/80 text-amber-950 hover:bg-amber-300'
                          : 'bg-red-500 text-white hover:bg-red-600';

                      const isSelected =
                        selectedHeatmapCell?.routeId === route.id &&
                        selectedHeatmapCell?.hour === cell.hour;

                      return (
                        <td key={cell.hour} className="p-0.5">
                          <button
                            type="button"
                            onClick={() => {
                              setSelectedHour(cell.hour);
                              setSelectedHeatmapCell({
                                routeId: route.id,
                                routeNumber: route.routeNumber,
                                routeName: route.name,
                                hour: cell.hour,
                                peakLoad: cell.peakLoad,
                                avgLoad: cell.avgLoad,
                                avgSpeed: cell.avgSpeed,
                                sampleTrips: cell.sampleTrips,
                              });
                            }}
                            title={`Route ${route.routeNumber} at ${String(cell.hour).padStart(
                              2,
                              '0'
                            )}:00 — ${cell.peakLoad}% (${cell.level}), ${cell.sampleTrips} trips (${cell.confidence} conf.)`}
                            className={`w-full h-7 min-w-[24px] rounded-xs font-mono text-[10px] font-medium tabular-nums transition-colors ${bg} ${
                              isSelected ? 'ring-2 ring-slate-900' : ''
                            }`}
                          >
                            {cell.peakLoad}
                          </button>
                        </td>
                      );
                    })}
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </section>

        <section className="xl:col-span-5 bg-white border border-slate-200 rounded-lg p-5">
          <div className="flex items-start justify-between gap-3 mb-4">
            <div>
              <h3 className="text-base font-bold text-slate-900">
                Route {activeRouteForStopProfile.routeNumber} Stretch Load
              </h3>
              <p className="text-xs text-slate-500 mt-0.5">
                {heatmapDayType === 'weekday' ? 'Weekday' : 'Weekend'} at{' '}
                <span className="font-mono font-semibold text-slate-700 tabular-nums">
                  {String(selectedHeatmapCell?.hour ?? selectedHour).padStart(2, '0')}:00
                </span>{' '}
                · {activeRouteForStopProfile.busSize} ({activeRouteForStopProfile.crushCapacity} cap)
              </p>
            </div>
            <button
              type="button"
              onClick={() => {
                const stops = activeRouteForStopProfile.stopIds;
                const hr = selectedHeatmapCell?.hour ?? selectedHour;
                onInspectCorridorInPlanner(
                  stops[0],
                  stops[stops.length - 1],
                  `${String(hr).padStart(2, '0')}:00`,
                  heatmapDayType
                );
              }}
              className="px-3 py-1.5 text-xs font-semibold text-blue-700 bg-blue-50 hover:bg-blue-100 rounded-md transition-colors whitespace-nowrap shrink-0"
            >
              Plan this route
            </button>
          </div>

          <div className="space-y-2.5">
            {stopProfileRecords.map((rec) => {
              const barColor =
                rec.crowdingLevel === 'Seats available'
                  ? 'bg-emerald-500'
                  : rec.crowdingLevel === 'Standing room'
                  ? 'bg-amber-500'
                  : 'bg-red-500';
              return (
                <div key={rec.id} className="space-y-1">
                  <div className="flex items-center justify-between text-xs">
                    <span className="font-medium text-slate-800 truncate pr-2">
                      {rec.stopSequence}. {rec.stopName}
                    </span>
                    <span className="font-mono text-slate-600 tabular-nums whitespace-nowrap shrink-0">
                      {rec.cumulativeLoad}/{rec.crushCapacity} pax ·{' '}
                      <strong className="text-slate-900">{rec.loadPercentage}%</strong>
                    </span>
                  </div>
                  <div className="w-full h-2 bg-slate-100 rounded-sm overflow-hidden">
                    <div
                      className={`h-full ${barColor} transition-all duration-300`}
                      style={{ width: `${rec.loadPercentage}%` }}
                    />
                  </div>
                </div>
              );
            })}
          </div>

          {stopProfileRecords[0] && (
            <div className="mt-4 pt-3 border-t border-slate-100 flex items-center justify-between text-xs text-slate-500 tabular-nums">
              <span>
                Sample size: {stopProfileRecords[0].sampleTrips} trips ({stopProfileRecords[0].confidence} confidence)
              </span>
              <span>Uncertainty: ±{stopProfileRecords[0].alightingUncertaintyPax} pax</span>
            </div>
          )}
        </section>
      </div>
    </div>
  );
}

function WeekdayVsWeekendChart({
  data,
  selectedHour,
  onHoverHour,
  onSelectHour,
}: {
  data: ReturnType<typeof buildHourlyDemandSeries>;
  selectedHour: number;
  onHoverHour: (hour: number | null) => void;
  onSelectHour: (hour: number) => void;
}) {
  const W = 920;
  const H = 255;
  const padL = 46;
  const padR = 46;
  const padT = 20;
  const padB = 34;

  const maxVol = Math.max(100, ...data.map((d) => d.totalVolume)) * 1.12;
  const chartW = W - padL - padR;
  const chartH = H - padT - padB;
  const slotW = chartW / Math.max(1, data.length);
  const barW = Math.min(26, slotW * 0.56);

  const yPct = (pct: number) => padT + chartH - (pct / 100) * chartH;

  const weekdayLine = data
    .map((d, idx) => {
      const x = padL + idx * slotW + slotW / 2;
      const y = yPct(d.avgLoadPercentage);
      return `${idx === 0 ? 'M' : 'L'} ${x.toFixed(1)},${y.toFixed(1)}`;
    })
    .join(' ');

  const weekendLine = data
    .map((d, idx) => {
      const x = padL + idx * slotW + slotW / 2;
      const y = yPct(d.weekendAvgLoadPercentage);
      return `${idx === 0 ? 'M' : 'L'} ${x.toFixed(1)},${y.toFixed(1)}`;
    })
    .join(' ');

  return (
    <div className="overflow-x-auto">
      <svg
        viewBox={`0 0 ${W} ${H}`}
        className="w-full min-w-[640px] h-64 block"
        role="img"
        aria-label="Weekday vs Weekend bus load and ticket demand comparison"
      >
        {[0, 25, 55, 85, 100].map((pct) => (
          <g key={pct}>
            <line
              x1={padL}
              y1={yPct(pct)}
              x2={W - padR}
              y2={yPct(pct)}
              stroke="#E2E8F0"
              strokeDasharray={pct === 85 || pct === 55 ? '4 4' : undefined}
            />
            <text
              x={W - padR + 6}
              y={yPct(pct) + 3}
              fontSize="9.5"
              fontFamily="JetBrains Mono, monospace"
              fill="#64748B"
            >
              {pct}%
            </text>
          </g>
        ))}

        {data.map((d, idx) => {
          const cx = padL + idx * slotW + slotW / 2;
          const x = cx - barW / 2;
          const totalH = (d.totalVolume / maxVol) * chartH;
          const isSelected = d.hour === selectedHour;

          return (
            <g
              key={d.hour}
              className="cursor-pointer"
              onMouseEnter={() => onHoverHour(d.hour)}
              onMouseLeave={() => onHoverHour(null)}
              onClick={() => onSelectHour(d.hour)}
            >
              {isSelected && (
                <rect
                  x={cx - slotW / 2 + 2}
                  y={padT}
                  width={slotW - 4}
                  height={chartH}
                  fill="#EFF6FF"
                  rx={4}
                />
              )}
              <rect
                x={x}
                y={padT + chartH - totalH}
                width={barW}
                height={totalH}
                fill="#CBD5E1"
                rx={1}
              />
              <text
                x={cx}
                y={H - 12}
                textAnchor="middle"
                fontSize="9.5"
                fontWeight={isSelected ? '700' : '500'}
                fontFamily="JetBrains Mono, monospace"
                fill={isSelected ? '#1D4ED8' : '#64748B'}
              >
                {String(d.hour).padStart(2, '0')}
              </text>
            </g>
          );
        })}

        <path
          d={weekendLine}
          fill="none"
          stroke="#0D9488"
          strokeWidth="2.2"
          strokeDasharray="5 4"
          strokeLinecap="round"
        />
        <path
          d={weekdayLine}
          fill="none"
          stroke="#DC2626"
          strokeWidth="2.4"
          strokeLinecap="round"
        />
        {data.map((d, idx) => {
          const cx = padL + idx * slotW + slotW / 2;
          return (
            <g key={`pts-${d.hour}`}>
              <circle
                cx={cx}
                cy={yPct(d.avgLoadPercentage)}
                r={d.hour === selectedHour ? 4.5 : 2.8}
                fill="#DC2626"
                stroke="#FFFFFF"
                strokeWidth="1.2"
              />
              <circle
                cx={cx}
                cy={yPct(d.weekendAvgLoadPercentage)}
                r={d.hour === selectedHour ? 4 : 2.4}
                fill="#0D9488"
                stroke="#FFFFFF"
                strokeWidth="1.2"
              />
            </g>
          );
        })}
      </svg>

      <div className="flex flex-wrap items-center justify-end gap-5 text-xs text-slate-600 mt-1">
        <span className="inline-flex items-center gap-1.5">
          <span className="w-3 h-2.5 bg-slate-300 rounded-xs" />
          Ticket Volume (ETM + QR)
        </span>
        <span className="inline-flex items-center gap-1.5">
          <span className="w-3.5 h-0.5 bg-red-600" />
          Weekday Mean Load (%)
        </span>
        <span className="inline-flex items-center gap-1.5">
          <span className="w-3.5 h-0.5 bg-teal-600" />
          Weekend Mean Load (%)
        </span>
      </div>
    </div>
  );
}
